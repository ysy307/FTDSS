    subroutine compute_time_derivative_variable(self, bdf_coeffs, bdf_order)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: bdf_coeffs(:)
        integer(int32), intent(in) :: bdf_order

        integer(int32) :: i, idx
        real(real64) :: coeff

        call self%assert_initialized(__LINE__, "types_physics_variable:compute_time_derivative_variable")

        if (size(bdf_coeffs) < bdf_order + 1) then
            error stop "compute_time_derivative: coeff size mismatch"
        end if

        self%diff = 0.0d0
        idx = self%head_idx

        do i = 1, bdf_order + 1
            coeff = bdf_coeffs(i)
            self%diff(:) = self%diff(:) + coeff * self%values(:, idx)
            idx = self%get_prev_index(idx)
        end do
    end subroutine compute_time_derivative_variable
