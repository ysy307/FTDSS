    module subroutine reset_params_hcf(self)
        implicit none
        class(type_hcf_params), intent(inout) :: self

        self%model_number = 0
        self%unit_id = 0
        self%water_viscosity_model = 0
        self%hcf_model_number = 0
        self%k_s = 0.0d0
        self%theta_s = 0.0d0
        self%theta_r = 0.0d0
        self%alpha1 = 0.0d0
        self%n1 = 0.0d0
        self%m1 = 0.0d0
        self%h_crit = 0.0d0
        self%alpha2 = 0.0d0
        self%n2 = 0.0d0
        self%m2 = 0.0d0
        self%w1 = 0.0d0
        self%w2 = 0.0d0
        self%l = 0.0d0
        self%omega = 0.0d0
        self%gain_factor = 0.0d0

    end subroutine reset_params_hcf
