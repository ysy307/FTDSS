    subroutine assert_initialized(self, line, scope)
        implicit none
        class(type_variable), intent(in) :: self
        character(len=*), intent(in) :: scope
        integer(int32), intent(in) :: line

        if (.not. self%is_initialized) then
            call raise_error(ERROR_CODES%NOT_INITIALIZED, &
                             opt="type_variable", &
                             scope=scope, &
                             line=line)
        end if
    end subroutine assert_initialized
