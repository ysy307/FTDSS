    subroutine reset_physics_info(self)
        implicit none
        class(type_physics_info), intent(inout) :: self

        self%num_phases = 0
        self%solid = 0.0d0
        self%water = 0.0d0
        self%ice = 0.0d0
        self%vapor = 0.0d0

        call deallocate_array(self%dispersivity)
        call deallocate_array(self%params)

    end subroutine reset_physics_info
