private subroutine unique_int16(array, unique_array)

Finds the unique elements in an int16 array and returns them in sorted order.

Arguments

Type IntentOptional Attributes Name
integer(kind=int16), intent(in) :: array(:)

The input array of int16 integers.
integer(kind=int16), intent(inout), allocatable :: unique_array(:)

An allocatable output array that will contain the sorted unique elements.
