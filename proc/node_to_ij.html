    pure subroutine node_to_ij(node, ix, iy)
        implicit none
        integer(int32), intent(in) :: node
        integer(int32), intent(inout) :: ix, iy

        select case (node)
        ! Vertices
        case (1);  ix = 1; iy = 1
        case (2);  ix = 4; iy = 1
        case (3);  ix = 4; iy = 4
        case (4);  ix = 1; iy = 4
        ! Edge 1-2
        case (5);  ix = 2; iy = 1
        case (6);  ix = 3; iy = 1
        ! Edge 2-3
        case (7);  ix = 4; iy = 2
        case (8);  ix = 4; iy = 3
        ! Edge 3-4
        case (9);  ix = 3; iy = 4
        case (10); ix = 2; iy = 4
        ! Edge 4-1
        case (11); ix = 1; iy = 3
        case (12); ix = 1; iy = 2
        ! Interior
        case (13); ix = 2; iy = 2
        case (14); ix = 3; iy = 2
        case (15); ix = 3; iy = 3
        case (16); ix = 2; iy = 3
        case default
            ix = 1; iy = 1
        end select
    end subroutine node_to_ij
