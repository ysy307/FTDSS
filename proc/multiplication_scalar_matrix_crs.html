    function Multiplication_Scalar_Matrix_CRS(A, b) result(C)
        implicit none
        type(Type_CRS), intent(in) :: A
        real(real64), intent(in) :: b
        type(Type_CRS) :: C
        integer(int32) :: k

        ! Assume same sparsity structure
        C = A%Copy()
        do k = 1, A%nnz
            C%Val(k) = A%Val(k) * b
        end do
    end function Multiplication_Scalar_Matrix_CRS
