    pure function block_fnorm_iluc(blk, bs) result(nrm)
        implicit none
        integer(int32), intent(in) :: bs
        real(real64), intent(in) :: blk(bs, bs)
        real(real64) :: nrm
        integer(int32) :: ii, jj
        nrm = 0.0d0
        do jj = 1, bs
            do ii = 1, bs
                nrm = nrm + blk(ii, jj)**2
            end do
        end do
        nrm = sqrt(nrm)
    end function block_fnorm_iluc
