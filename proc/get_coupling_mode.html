    pure function get_coupling_mode(key) result(val)
        implicit none
        character(len=*), intent(in) :: key
        integer(int32) :: val

        select case (strip(to_lower(key)))
        case ("weak")
            val = COUPLING_MODE_STAGGERED
        case ("strong")
            val = COUPLING_MODE_MONOLITHIC
        end select
    end function get_coupling_mode
