    subroutine set_previous_scalar_all_variable(self, val)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: val
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "types_physics_variable:set_previous_scalar_all_variable")

        prev = self%get_prev_index(self%head_idx)
        self%values(:, prev) = val
    end subroutine set_previous_scalar_all_variable
