    pure subroutine get_legendre_point(n, idx, w, p)
        implicit none
        integer(int32), intent(in) :: n
        integer(int32), intent(in) :: idx
        real(real64), intent(inout) :: w, p

        real(real64), parameter :: s3 = sqrt(1.0d0 / 3.0d0)
        real(real64), parameter :: s35 = sqrt(3.0d0 / 5.0d0)

        select case (n)
        case (1)
            p = 0.0d0
            w = 2.0d0
        case (2)
            if (idx == 1) then
                p = -s3
                w = 1.0d0
            else
                p = s3
                w = 1.0d0
            end if
        case (3)
            select case (idx)
            case (1)
                p = -s35
                w = 5.0d0 / 9.0d0
            case (2)
                p = 0.0d0
                w = 8.0d0 / 9.0d0
            case (3)
                p = s35
                w = 5.0d0 / 9.0d0
            case default
                p = 0.0d0
                w = 0.0d0
            end select
        case default
            ! Fallback
            p = 0.0d0
            w = 2.0d0
        end select
    end subroutine get_legendre_point
