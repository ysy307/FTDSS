private subroutine get_dt(self, dt, time_unit)

Get current time step size (seconds) with optional unit conversion. If a target time unit is provided, dt is converted accordingly.

    subroutine get_dt(self, dt, time_unit)
        implicit none
        class(type_time), intent(in) :: self
        !> Output time step
        real(real64), intent(inout) :: dt
        !> Target time unit (optional)
        type(type_constant_value), intent(in), optional :: time_unit

        real(real64) :: coeff

        if (present(time_unit)) then
            coeff = 1.0d0 / time_unit%value
        else
            coeff = 1.0d0
        end if

        dt = self%dt * coeff
    end subroutine get_dt
