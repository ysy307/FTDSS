    subroutine get_previous_scalar(self, node_id, val)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: node_id
        real(real64), intent(inout) :: val
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "core_types_variable:get_previous_scalar")

        if (node_id < 1 .or. node_id > self%num_dof) then
            error stop "get_previous: invalid node_id"
        end if

        prev = self%get_prev_index(self%head_idx)
        val = self%values(node_id, prev)
    end subroutine get_previous_scalar
