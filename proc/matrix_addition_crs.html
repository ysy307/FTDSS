    function Matrix_Addition_CRS(A, B) result(C)
        implicit none
        type(Type_CRS), intent(in) :: A, B
        type(Type_CRS) :: C
        integer(int32) :: k

        ! Assume same sparsity structure
        C = A%Copy()
        do k = 1, A%nnz
            C%Val(k) = A%Val(k) + B%Val(k)
        end do
    end function Matrix_Addition_CRS
