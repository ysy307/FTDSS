    subroutine zero_type_coordinate_array_dp(self)
        implicit none
        !> The coordinate array object to zero.
        class(type_coordinate_array_dp), intent(inout) :: self

        if (self%is_allocated) then
            self%x(:) = 0.0d0
            self%y(:) = 0.0d0
            self%z(:) = 0.0d0
        end if
    end subroutine zero_type_coordinate_array_dp
