    module subroutine destroy_abst_fe(self)
        implicit none
        class(abst_fe), intent(inout) :: self

        self%type = 0
        self%num_nodes = 0
        self%dimension = 0
        call self%integration_rule%destroy()
    end subroutine destroy_abst_fe
