    pure function logic_to_string(id) result(str)
        integer(int32), intent(in) :: id
        character(len=16) :: str
        str = trim(NONLINEAR_LOGIC%to_name(id))
        if (len_trim(str) == 0) str = "unknown"
    end function logic_to_string
