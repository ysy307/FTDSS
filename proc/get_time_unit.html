    pure function get_time_unit(key) result(val)
        implicit none
        character(len=*), intent(in) :: key
        integer(int32) :: val

        select case (strip(to_lower(key)))
        case ("second")
            val = TIME_UNIT_SECONDS
        case ("minute")
            val = TIME_UNIT_MINUTES
        case ("hour")
            val = TIME_UNIT_HOURS
        case ("day")
            val = TIME_UNIT_DAYS
        case ("year")
            val = TIME_UNIT_YEARS
        case default
            val = -1
        end select
    end function get_time_unit
