    pure function get_next_index(self, current_idx) result(next_val)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: current_idx
        integer(int32) :: next_val

        if (current_idx == self%num_time_slots) then
            next_val = 1
        else
            next_val = current_idx + 1
        end if
    end function get_next_index
