    module subroutine initialize_bc(self, config_bc)
        implicit none
        class(abst_bc), intent(inout) :: self
        type(type_config_bc), intent(in) :: config_bc

        call self%config%copy(config_bc)

        self%current_idx = 0

        self%initialized = .true.

    end subroutine initialize_bc
