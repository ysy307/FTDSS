    subroutine zero_all_jacobian_matrix(self)
        implicit none
        class(type_jacobian_matrix), intent(inout) :: self
        integer(int32) :: i

        if (allocated(self%matrix)) then
            do i = 1, size(self%matrix)
                call self%matrix(i)%zero()
            end do
        end if
    end subroutine zero_all_jacobian_matrix
