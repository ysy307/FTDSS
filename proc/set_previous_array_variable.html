    subroutine set_previous_array_variable(self, val)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: val(:)
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "core_types_variable:set_previous_array_variable")

        if (size(val) /= self%num_dof) then
            error stop "set_previous: size mismatch"
        end if

        prev = self%get_prev_index(self%head_idx)
        self%values(:, prev) = val
    end subroutine set_previous_array_variable
