    pure function get_prev_index(self, current_idx) result(prev)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: current_idx
        integer(int32) :: prev

        if (current_idx == 1) then
            prev = self%num_time_slots
        else
            prev = current_idx - 1
        end if
    end function get_prev_index
