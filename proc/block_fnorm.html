    pure function block_fnorm(blk, bs) result(nrm)
        implicit none
        integer(int32), intent(in) :: bs
        real(real64), intent(in) :: blk(bs, bs)
        real(real64) :: nrm
        integer(int32) :: i, j

        nrm = 0.0d0
        do j = 1, bs
            do i = 1, bs
                nrm = nrm + blk(i, j)**2
            end do
        end do
        nrm = sqrt(nrm)
    end function block_fnorm
