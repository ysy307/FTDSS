    subroutine initialize_ic(self, config_ic)
        implicit none
        class(abst_ic), intent(inout) :: self
        type(type_config_ic), intent(in) :: config_ic

        call self%config%copy(config_ic)
        self%initialized = .true.
    end subroutine initialize_ic
