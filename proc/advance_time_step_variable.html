    subroutine advance_time_step_variable(self)
        implicit none
        class(type_variable), intent(inout) :: self
        integer(int32) :: old_head

        call self%assert_initialized(__LINE__, "core_types_variable:advance_time_step_variable")

        old_head = self%head_idx
        self%head_idx = self%get_next_index(old_head)

        self%values(:, self%head_idx) = self%values(:, old_head)
    end subroutine advance_time_step_variable
