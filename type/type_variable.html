    type :: type_variable
        logical, private :: is_initialized = .false.

        integer(int32), private :: num_dof = 0
        integer(int32), private :: num_history_steps = 0
        integer(int32), private :: num_time_slots = 0
        integer(int32), private :: head_idx = 1

        real(real64), allocatable, private :: values(:, :)
        real(real64), allocatable, private :: delta(:)
        real(real64), allocatable, private :: diff(:)

        type(type_coordinate_array_dp), private :: grad

    contains

        procedure, public :: initialize => initialize_type_variable
        procedure, public :: destroy => destroy_type_variable

        procedure, public :: advance => advance_time_step_variable
        procedure, public :: restore => restore_previous_step_variable
        procedure, public :: reset => reset_all_states_variable

        procedure, private :: assert_initialized

        ! --- Current Setters ---
        procedure, private :: set_current_array_variable
        procedure, private :: set_current_scalar_variable
        procedure, private :: set_current_scalar_all_variable
        generic, public :: set_current => &
            set_current_array_variable, &
            set_current_scalar_variable, &
            set_current_scalar_all_variable

        ! --- Previous Setters ---
        procedure, private :: set_previous_array_variable
        procedure, private :: set_previous_scalar_variable
        procedure, private :: set_previous_scalar_all_variable
        generic, public :: set_previous => &
            set_previous_array_variable, &
            set_previous_scalar_variable, &
            set_previous_scalar_all_variable

        procedure, public :: set_delta_array_variable
        generic, public :: set_delta => set_delta_array_variable

        procedure, public :: get_delta
        procedure, public :: get_diff
        procedure, public :: get_history

        procedure, public :: get_current_gradient_array
        procedure, public :: get_current_gradient_scalar
        generic, public :: get_current_gradient => &
            get_current_gradient_array, &
            get_current_gradient_scalar

        ! --- Current Getters ---
        procedure, private :: get_current_values
        procedure, private :: get_current_scalar
        generic, public :: get_current => &
            get_current_values, &
            get_current_scalar

        ! --- Previous Getters ---
        procedure, private :: get_previous_values
        procedure, private :: get_previous_scalar
        generic, public :: get_previous => &
            get_previous_values, &
            get_previous_scalar

        procedure, public :: compute_time_derivative => compute_time_derivative_variable
        procedure, public :: get_head_index

        ! Helper functions for circular buffer indexing
        procedure, private :: get_prev_index
        procedure, private :: get_next_index

    end type type_variable
