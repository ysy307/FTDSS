module subroutine set_local_matrix_bsr(self, op, n_local, indices, row_sys, col_sys, local_data)

Scatter a local element matrix into BSR global matrix using pre-computed block indices. O(1) direct address scatter — no binary search. row_sys/col_sys: 1-based DOF offsets within the BSR block for sub-block scattering. op = MATRIX_OPS%INS overwrites; op = MATRIX_OPS%ADD accumulates.
