private module subroutine calc_effective_density_value_hydraulic(self, state, rho_eff)

Arguments

Type IntentOptional Attributes Name
class(type_hydraulic), intent(in) :: self
type(type_state), intent(in) :: state
real(kind=real64), intent(inout) :: rho_eff

Description

Evaluate the pore-water effective density at the supplied state. [kg/m3]. Plain (non-time-derivative) conserved storage quantity of the water-mass balance; phase contents must already be consistent with (T, p_w).
