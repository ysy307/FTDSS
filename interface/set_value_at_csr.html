private module subroutine set_value_at_csr(self, op, idx, value)

Arguments

Type IntentOptional Attributes Name
class(type_matrix_csr), intent(inout), target :: self
type(type_constant_id), intent(in) :: op
integer(kind=int32), intent(in) :: idx
real(kind=real64), intent(in) :: value

Description

Sets a single stored entry addressed by its pre-computed flat position in the contiguous value storage (no row/col search). op=INS overwrites; op=ADD accumulates.
