    subroutine deallocate_rank1_int8(array)
        implicit none
        integer(int8), allocatable, intent(inout) :: array(:)
        integer(int32) :: stat

        if (allocated(array)) then
            deallocate (array, stat=stat)
            if (stat /= 0) call error_message(971)
        end if
