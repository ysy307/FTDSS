#:set VTYPES = [('dp', 'real(real64)', 'real64'), ('int', 'integer(int32)', 'int32')]
!>
!> Abstract interface for double-precision and integer vector types.
!> Storage layout: val((node-1)*num_blocks + block), interleaved by block.
!>
module core_types_algebra_vector
    use, intrinsic :: iso_fortran_env
!$  use :: omp_lib
    use :: stdlib_optval, only:optval
    use :: core_constants
    use :: core_check_range, only:value_in_range
    use :: core_memory, only:allocate_array, deallocate_array
    implicit none
    private

    public :: type_vector_dp
    public :: type_vector_int

    ! ==========================================================
    #:for VTYPE, FTYPE, KTYPE in VTYPES
    ! ${VTYPE}$ Vector
    ! ==========================================================
    type :: type_vector_${VTYPE}$
        private
        ${FTYPE}$, allocatable :: val(:)
        integer(int32) :: num_nodes = 0
        integer(int32) :: num_blocks = 0
        logical :: is_initialized_vector = .false.
        type(type_constant_id) :: status = VECTOR_STATUS%SUCCESS
    contains
        ! ---- Lifecycle ----
        procedure, public, pass(self) :: initialize => initialize_vector_${VTYPE}$
        procedure, public, pass(self) :: destroy => destroy_vector_${VTYPE}$
        ! ---- Mutator ----
        #:if VTYPE == 'dp'
        procedure, private, pass(self) :: set_scalar => set_scalar_vector_${VTYPE}$
        #:endif
        procedure, private, pass(self) :: set_array => set_array_vector_${VTYPE}$
        procedure, private, pass(self) :: set_value_at_index => set_value_at_index_vector_${VTYPE}$
        procedure, private, pass(self) :: set_values_at_indices => set_values_at_indices_vector_${VTYPE}$
        #:if VTYPE == 'dp'
        generic, public :: set => set_scalar, set_array, set_value_at_index, set_values_at_indices
        #:else
        generic, public :: set => set_array, set_value_at_index, set_values_at_indices
        #:endif
        procedure, public, pass(self) :: scale => scale_vector_${VTYPE}$
        procedure, public, pass(self) :: copy => copy_vector_${VTYPE}$
        procedure, public, pass(self) :: zero => zero_vector_${VTYPE}$
        ! ---- Getter ----
        procedure, public, pass(self) :: get_size => get_size_vector_${VTYPE}$
        procedure, public, pass(self) :: get_data => get_data_vector_${VTYPE}$
        procedure, public, pass(self) :: get_status => get_status_vector_${VTYPE}$
        ! ---- Meta / Utility ----
        procedure, public, pass(self) :: display => display_vector_${VTYPE}$
        procedure, public, pass(self) :: check => check_vector_${VTYPE}$
        ! ---- Operator ----
        procedure, public, pass(self) :: is_initialized => is_initialized_vector_${VTYPE}$
    end type type_vector_${VTYPE}$

    #:endfor

    interface
        #:for VTYPE, FTYPE, KTYPE in VTYPES
        ! ---- ${VTYPE}$ ----
        module subroutine initialize_vector_${VTYPE}$(self, num_nodes, num_blocks)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
            integer(int32), intent(in) :: num_nodes
            integer(int32), intent(in), optional :: num_blocks
        end subroutine initialize_vector_${VTYPE}$

        module subroutine destroy_vector_${VTYPE}$(self)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
        end subroutine destroy_vector_${VTYPE}$

        module pure function is_initialized_vector_${VTYPE}$(self) result(initialized)
            implicit none
            class(type_vector_${VTYPE}$), intent(in) :: self
            logical :: initialized
        end function is_initialized_vector_${VTYPE}$

        module pure function get_size_vector_${VTYPE}$(self) result(vector_size)
            implicit none
            class(type_vector_${VTYPE}$), intent(in) :: self
            integer(int32) :: vector_size
        end function get_size_vector_${VTYPE}$

        module function get_data_vector_${VTYPE}$(self) result(data)
            implicit none
            class(type_vector_${VTYPE}$), intent(in), target :: self
            ${FTYPE}$, pointer, dimension(:) :: data
        end function get_data_vector_${VTYPE}$

        module pure function get_status_vector_${VTYPE}$(self) result(status)
            implicit none
            class(type_vector_${VTYPE}$), intent(in) :: self
            type(type_constant_id) :: status
        end function get_status_vector_${VTYPE}$

        #:if VTYPE == 'dp'
        module subroutine set_scalar_vector_${VTYPE}$(self, op, scalar_value, row_block)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
            type(type_constant_id), intent(in) :: op
            ${FTYPE}$, intent(in) :: scalar_value
            integer(int32), intent(in), optional :: row_block
        end subroutine set_scalar_vector_${VTYPE}$

        #:endif
        module subroutine set_array_vector_${VTYPE}$(self, op, array_value, row_block)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
            type(type_constant_id), intent(in) :: op
            ${FTYPE}$, intent(in) :: array_value(:)
            integer(int32), intent(in), optional :: row_block
        end subroutine set_array_vector_${VTYPE}$

        module subroutine set_value_at_index_vector_${VTYPE}$(self, op, global_index, value, row_block)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
            type(type_constant_id), intent(in) :: op
            integer(int32), intent(in) :: global_index
            ${FTYPE}$, intent(in) :: value
            integer(int32), intent(in), optional :: row_block
        end subroutine set_value_at_index_vector_${VTYPE}$

        module subroutine set_values_at_indices_vector_${VTYPE}$(self, op, global_indices, new_values, row_block)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
            type(type_constant_id), intent(in) :: op
            integer(int32), intent(in) :: global_indices(:)
            ${FTYPE}$, intent(in) :: new_values(:)
            integer(int32), intent(in), optional :: row_block
        end subroutine set_values_at_indices_vector_${VTYPE}$

        module subroutine scale_vector_${VTYPE}$(self, op, alpha)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
            type(type_constant_id), intent(in) :: op
            class(type_vector_${VTYPE}$), intent(in) :: alpha
        end subroutine scale_vector_${VTYPE}$

        module subroutine copy_vector_${VTYPE}$(self, source_vector)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
            class(type_vector_${VTYPE}$), intent(in) :: source_vector
        end subroutine copy_vector_${VTYPE}$

        module subroutine zero_vector_${VTYPE}$(self)
            implicit none
            class(type_vector_${VTYPE}$), intent(inout) :: self
        end subroutine zero_vector_${VTYPE}$

        module subroutine display_vector_${VTYPE}$(self, unit_in)
            implicit none
            class(type_vector_${VTYPE}$), intent(in) :: self
            integer(int32), intent(in), optional :: unit_in
        end subroutine display_vector_${VTYPE}$

        module subroutine check_vector_${VTYPE}$(self)
            implicit none
            class(type_vector_${VTYPE}$), intent(in) :: self
        end subroutine check_vector_${VTYPE}$

        #:endfor
    end interface

end module core_types_algebra_vector
