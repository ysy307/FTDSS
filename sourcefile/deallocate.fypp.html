#:include "common.fypp"

!> Module providing generic deallocation routines with error handling.
!> Uses fypp preprocessor to generate procedures for various types and ranks.
module core_deallocate
    use, intrinsic :: iso_fortran_env
    use :: core_constants, only:ERROR_CODES
    use :: core_error, only:raise_error
    implicit none
    private

    public :: deallocate_array

    !> Generic interface for safe array deallocation.
    interface deallocate_array
        #:for rank in range(1, MAX_RANK + 1)
            ! Rank-${rank}$
            #:for dtype in ['integer', 'real', 'logical']
                #:for k in KINDS[dtype]
                    #:set suffix = ('logical' + k) if dtype == 'logical' else k
                    module procedure :: deallocate_rank${rank}$_${suffix}$
                #:endfor
            #:endfor
        #:endfor
    end interface

contains

    #:for rank in range(1, MAX_RANK + 1)
        #:for dtype in ['integer', 'real', 'logical']
            #:for k, t_decl in zip(KINDS[dtype], TYPES[dtype])
                #:set suffix = ('logical' + k) if dtype == 'logical' else k
                ! ==========================================================
                ! Rank-${rank}$, Type: ${t_decl}$
                ! ==========================================================

                !> Safely deallocate a Rank-${rank}$ array of type ${t_decl}$.
                subroutine deallocate_rank${rank}$_${suffix}$ (array)
                    implicit none
                    !> The array to be deallocated
                    ${t_decl}$, allocatable, intent(inout) :: array${rank_suffix(rank)}$
                    integer(int32) :: stat

                    if (allocated(array)) then
                        deallocate (array, stat=stat)
                        if (stat /= 0) call raise_error(ERROR_CODES%DEALLOC_FAILED, &
                                                        scope="core_deallocate:deallocate_rank${rank}$_${suffix}$", &
                                                        line=__LINE__)
                    end if
                end subroutine deallocate_rank${rank}$_${suffix}$

            #:endfor
        #:endfor
    #:endfor

end module core_deallocate
