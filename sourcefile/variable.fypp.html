module types_physics_variable
    use, intrinsic :: iso_fortran_env, only: real64, int32
    use :: core_constants, only:ERROR_CODES
    use :: core_memory
    use :: core_validation, only:raise_error
    use :: core_types_geometry_coordinate_array, only:type_coordinate_array_dp
    use :: core_types_geometry_coordinate, only:type_coordinate_dp
    implicit none
    private

    public :: type_variable

    type :: type_variable
        logical, private :: is_initialized = .false.

        integer(int32), private :: num_dof = 0
        integer(int32), private :: num_history_steps = 0
        integer(int32), private :: num_time_slots = 0
        integer(int32), private :: head_idx = 1

        real(real64), allocatable, private :: values(:, :)
        real(real64), allocatable, private :: delta(:)
        real(real64), allocatable, private :: diff(:)

        type(type_coordinate_array_dp), private :: grad

    contains

        procedure, public :: initialize => initialize_type_variable
        procedure, public :: destroy => destroy_type_variable

        procedure, public :: advance => advance_time_step_variable
        procedure, public :: restore => restore_previous_step_variable
        procedure, public :: reset => reset_all_states_variable

        procedure, private :: assert_initialized

        ! --- Current Setters ---
        procedure, private :: set_current_array_variable
        procedure, private :: set_current_scalar_variable
        procedure, private :: set_current_scalar_all_variable
        generic, public :: set_current => &
            set_current_array_variable, &
            set_current_scalar_variable, &
            set_current_scalar_all_variable

        ! --- Previous Setters ---
        procedure, private :: set_previous_array_variable
        procedure, private :: set_previous_scalar_variable
        procedure, private :: set_previous_scalar_all_variable
        generic, public :: set_previous => &
            set_previous_array_variable, &
            set_previous_scalar_variable, &
            set_previous_scalar_all_variable

        procedure, public :: set_delta_array_variable
        generic, public :: set_delta => set_delta_array_variable

        procedure, public :: get_delta
        procedure, public :: get_diff
        procedure, public :: get_history

        procedure, public :: get_current_gradient_array
        procedure, public :: get_current_gradient_scalar
        generic, public :: get_current_gradient => &
            get_current_gradient_array, &
            get_current_gradient_scalar

        ! --- Current Getters ---
        procedure, private :: get_current_values
        procedure, private :: get_current_scalar
        generic, public :: get_current => &
            get_current_values, &
            get_current_scalar

        ! --- Previous Getters ---
        procedure, private :: get_previous_values
        procedure, private :: get_previous_scalar
        generic, public :: get_previous => &
            get_previous_values, &
            get_previous_scalar

        procedure, public :: compute_time_derivative => compute_time_derivative_variable
        procedure, public :: get_head_index

        ! Helper functions for circular buffer indexing
        procedure, private :: get_prev_index
        procedure, private :: get_next_index

    end type type_variable

contains

    subroutine assert_initialized(self, line, scope)
        implicit none
        class(type_variable), intent(in) :: self
        character(len=*), intent(in) :: scope
        integer(int32), intent(in) :: line

        if (.not. self%is_initialized) then
            call raise_error(ERROR_CODES%NOT_INITIALIZED, &
                             opt="type_variable", &
                             scope=scope, &
                             line=line)
        end if
    end subroutine assert_initialized

    subroutine initialize_type_variable(self, num_dof, num_history_steps)
        implicit none
        class(type_variable), intent(inout) :: self
        integer(int32), intent(in) :: num_dof
        integer(int32), intent(in) :: num_history_steps

        call self%destroy()

        self%num_dof = num_dof
        self%num_history_steps = num_history_steps
        self%num_time_slots = num_history_steps + 2
        self%head_idx = 1

        allocate (self%values(num_dof, self%num_time_slots))
        allocate (self%delta(num_dof))
        allocate (self%diff(num_dof))

        call self%grad%initialize(num_dof, 0.0d0)

        self%values = 0.0d0
        self%delta = 0.0d0
        self%diff = 0.0d0

        self%is_initialized = .true.
    end subroutine initialize_type_variable

    subroutine destroy_type_variable(self)
        implicit none
        class(type_variable), intent(inout) :: self

        if (self%is_initialized) then
            if (allocated(self%values)) deallocate (self%values)
            if (allocated(self%delta)) deallocate (self%delta)
            if (allocated(self%diff)) deallocate (self%diff)
            call self%grad%destroy()
            self%is_initialized = .false.
        end if
    end subroutine destroy_type_variable

    subroutine advance_time_step_variable(self)
        implicit none
        class(type_variable), intent(inout) :: self
        integer(int32) :: old_head

        call self%assert_initialized(__LINE__, "types_physics_variable:advance_time_step_variable")

        old_head = self%head_idx
        self%head_idx = self%get_next_index(old_head)

        self%values(:, self%head_idx) = self%values(:, old_head)
    end subroutine advance_time_step_variable

    subroutine restore_previous_step_variable(self)
        implicit none
        class(type_variable), intent(inout) :: self
        call self%assert_initialized(__LINE__, "types_physics_variable:restore_previous_step_variable")
        self%head_idx = self%get_prev_index(self%head_idx)
    end subroutine restore_previous_step_variable

    subroutine reset_all_states_variable(self, initial_value)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: initial_value(:)
        integer(int32) :: j

        call self%assert_initialized(__LINE__, "types_physics_variable:reset_all_states_variable")

        if (size(initial_value) /= self%num_dof) then
            error stop "reset: size mismatch"
        end if

        do j = 1, self%num_time_slots
            self%values(:, j) = initial_value
        end do

        self%delta = 0.0d0
        self%diff = 0.0d0
        call self%grad%zero()

        self%head_idx = 1
    end subroutine reset_all_states_variable

    subroutine get_history(self, node_id, out_array)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: node_id
        real(real64), intent(inout) :: out_array(:)

        integer(int32) :: i, idx, ncopy

        call self%assert_initialized(__LINE__, "types_physics_variable:get_history")

        if (node_id < 1 .or. node_id > self%num_dof) then
            error stop "get_history: invalid node_id"
        end if

        ncopy = min(int(size(out_array), int32), self%num_time_slots)
        idx = self%head_idx

        do i = 1, ncopy
            out_array(i) = self%values(node_id, idx)
            idx = self%get_prev_index(idx)
        end do
    end subroutine get_history

    ! --- Current Getters ---

    subroutine get_current_values(self, values)
        implicit none
        class(type_variable), intent(inout), target :: self
        real(real64), pointer, intent(inout), contiguous :: values(:)

        call self%assert_initialized(__LINE__, "types_physics_variable:get_current_values")
        values => self%values(:, self%head_idx)
    end subroutine get_current_values

    subroutine get_current_scalar(self, node_id, val)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: node_id
        real(real64), intent(inout) :: val

        call self%assert_initialized(__LINE__, "types_physics_variable:get_current_scalar")

        if (node_id < 1 .or. node_id > self%num_dof) then
            error stop "get_current: invalid node_id"
        end if

        val = self%values(node_id, self%head_idx)
    end subroutine get_current_scalar

    ! --- Previous Getters ---

    subroutine get_previous_values(self, values)
        implicit none
        class(type_variable), intent(inout), target :: self
        real(real64), pointer, intent(inout), contiguous :: values(:)
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "types_physics_variable:get_previous_values")
        prev = self%get_prev_index(self%head_idx)
        values => self%values(:, prev)
    end subroutine get_previous_values

    subroutine get_previous_scalar(self, node_id, val)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: node_id
        real(real64), intent(inout) :: val
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "types_physics_variable:get_previous_scalar")

        if (node_id < 1 .or. node_id > self%num_dof) then
            error stop "get_previous: invalid node_id"
        end if

        prev = self%get_prev_index(self%head_idx)
        val = self%values(node_id, prev)
    end subroutine get_previous_scalar

    ! -----------------------

    subroutine get_delta(self, values)
        implicit none
        class(type_variable), intent(inout), target :: self
        real(real64), pointer, intent(inout), contiguous :: values(:)

        call self%assert_initialized(__LINE__, "types_physics_variable:get_delta")
        values => self%delta
    end subroutine get_delta

    subroutine get_diff(self, values)
        implicit none
        class(type_variable), intent(inout), target :: self
        real(real64), pointer, intent(inout), contiguous :: values(:)

        call self%assert_initialized(__LINE__, "types_physics_variable:get_diff")
        values => self%diff
    end subroutine get_diff

    subroutine get_current_gradient_array(self, values)
        implicit none
        class(type_variable), intent(inout), target :: self
        type(type_coordinate_array_dp), pointer, intent(inout) :: values

        call self%assert_initialized(__LINE__, "types_physics_variable:get_current_gradient_array")
        values => self%grad
    end subroutine get_current_gradient_array

    subroutine get_current_gradient_scalar(self, node_id, grad)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: node_id
        type(type_coordinate_dp), intent(inout) :: grad

        call self%assert_initialized(__LINE__, "types_physics_variable:get_current_gradient_scalar")

        if (node_id < 1 .or. node_id > self%num_dof) then
            error stop "get_current_gradient: invalid node_id"
        end if

        call self%grad%get_coordinate(node_id, grad)
    end subroutine get_current_gradient_scalar

    ! --- Set Current Implementations ---

    subroutine set_current_array_variable(self, val)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: val(:)

        call self%assert_initialized(__LINE__, "types_physics_variable:set_current_array_variable")

        if (size(val) /= self%num_dof) then
            error stop "set_current: size mismatch"
        end if

        self%values(:, self%head_idx) = val
    end subroutine set_current_array_variable

    subroutine set_current_scalar_variable(self, node_id, val)
        implicit none
        class(type_variable), intent(inout) :: self
        integer(int32), intent(in) :: node_id
        real(real64), intent(in) :: val

        call self%assert_initialized(__LINE__, "types_physics_variable:set_current_scalar_variable")

        if (node_id < 1 .or. node_id > self%num_dof) then
            error stop "set_current_scalar: invalid node_id"
        end if

        self%values(node_id, self%head_idx) = val
    end subroutine set_current_scalar_variable

    subroutine set_current_scalar_all_variable(self, val)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: val

        call self%assert_initialized(__LINE__, "types_physics_variable:set_current_scalar_all_variable")
        self%values(:, self%head_idx) = val
    end subroutine set_current_scalar_all_variable

    ! --- Set Previous Implementations ---

    subroutine set_previous_array_variable(self, val)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: val(:)
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "types_physics_variable:set_previous_array_variable")

        if (size(val) /= self%num_dof) then
            error stop "set_previous: size mismatch"
        end if

        prev = self%get_prev_index(self%head_idx)
        self%values(:, prev) = val
    end subroutine set_previous_array_variable

    subroutine set_previous_scalar_variable(self, node_id, val)
        implicit none
        class(type_variable), intent(inout) :: self
        integer(int32), intent(in) :: node_id
        real(real64), intent(in) :: val
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "types_physics_variable:set_previous_scalar_variable")

        if (node_id < 1 .or. node_id > self%num_dof) then
            error stop "set_previous_scalar: invalid node_id"
        end if

        prev = self%get_prev_index(self%head_idx)
        self%values(node_id, prev) = val
    end subroutine set_previous_scalar_variable

    subroutine set_previous_scalar_all_variable(self, val)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: val
        integer(int32) :: prev

        call self%assert_initialized(__LINE__, "types_physics_variable:set_previous_scalar_all_variable")

        prev = self%get_prev_index(self%head_idx)
        self%values(:, prev) = val
    end subroutine set_previous_scalar_all_variable

    subroutine set_delta_array_variable(self, val)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: val(:)

        call self%assert_initialized(__LINE__, "types_physics_variable:set_delta_array_variable")

        if (size(val) /= self%num_dof) then
            error stop "set_delta: size mismatch"
        end if

        self%delta = val
        self%values(:, self%head_idx) = self%values(:, self%head_idx) + val
    end subroutine set_delta_array_variable

    subroutine compute_time_derivative_variable(self, bdf_coeffs, bdf_order)
        implicit none
        class(type_variable), intent(inout) :: self
        real(real64), intent(in) :: bdf_coeffs(:)
        integer(int32), intent(in) :: bdf_order

        integer(int32) :: i, idx
        real(real64) :: coeff

        call self%assert_initialized(__LINE__, "types_physics_variable:compute_time_derivative_variable")

        if (size(bdf_coeffs) < bdf_order + 1) then
            error stop "compute_time_derivative: coeff size mismatch"
        end if

        self%diff = 0.0d0
        idx = self%head_idx

        do i = 1, bdf_order + 1
            coeff = bdf_coeffs(i)
            self%diff(:) = self%diff(:) + coeff * self%values(:, idx)
            idx = self%get_prev_index(idx)
        end do
    end subroutine compute_time_derivative_variable

    pure function get_head_index(self) result(idx)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32) :: idx
        idx = self%head_idx
    end function get_head_index

    ! --- Private Helper Functions for Circular Buffer ---

    pure function get_prev_index(self, current_idx) result(prev)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: current_idx
        integer(int32) :: prev

        if (current_idx == 1) then
            prev = self%num_time_slots
        else
            prev = current_idx - 1
        end if
    end function get_prev_index

    pure function get_next_index(self, current_idx) result(next_val)
        implicit none
        class(type_variable), intent(in) :: self
        integer(int32), intent(in) :: current_idx
        integer(int32) :: next_val

        if (current_idx == self%num_time_slots) then
            next_val = 1
        else
            next_val = current_idx + 1
        end if
    end function get_next_index

end module types_physics_variable
